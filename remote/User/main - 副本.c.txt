/* --- 全局变量 --- */
static volatile uint32_t msTi = 0;

/* --- 时间管理函数 --- */

void TIM_Config(TIM_TypeDef* TIMx, uint16_t AutoReload, uint16_t Prescaler) {
    TIM_TimeBaseInitTypeDef TIM_TimeBaseStructure;
    NVIC_InitTypeDef NVIC_InitStructure;

    // 1. 开启时钟
    if (TIMx == TIM2) RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM2, ENABLE);
    else if (TIMx == TIM3) RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM3, ENABLE);
    else if (TIMx == TIM4) RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM4, ENABLE);

    // 2. 时基配置
    TIM_TimeBaseStructure.TIM_Period = AutoReload - 1;      
    TIM_TimeBaseStructure.TIM_Prescaler = Prescaler - 1; 
    TIM_TimeBaseStructure.TIM_ClockDivision = 0;
    TIM_TimeBaseStructure.TIM_CounterMode = TIM_CounterMode_Up; 
    TIM_TimeBaseInit(TIMx, &TIM_TimeBaseStructure);

    // 3. 开启更新中断
    TIM_ITConfig(TIMx, TIM_IT_Update, ENABLE);

    // 4. NVIC 配置
    if (TIMx == TIM2) NVIC_InitStructure.NVIC_IRQChannel = TIM2_IRQn;
    else if (TIMx == TIM3) NVIC_InitStructure.NVIC_IRQChannel = TIM3_IRQn;
    else if (TIMx == TIM4) NVIC_InitStructure.NVIC_IRQChannel = TIM4_IRQn;
    
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);

    // 5. 使能定时器
    TIM_Cmd(TIMx, ENABLE);
}

// TIM3 中断服务函数 (已修复: 包含清除标志位)
void TIM3_IRQHandler(void) {
    if (TIM_GetITStatus(TIM3, TIM_IT_Update) != RESET) {
        TIM_ClearITPendingBit(TIM3, TIM_IT_Update); // 必须清除!
        msTi++;
    }
}

// 获取系统运行毫秒数
uint32_t GetTick(void) {
    return msTi;
}

// 假设我们把被占用的按键 PA2/PA3 移到了 PA6/PA7
// 新的按键分配: PA0, PA1, PA4, PA6, PA7

/* --- DMA 配置 (针对 USART2_TX 使用 DMA1_Channel7) --- */
void USART2_DMA_Configuration(void) {
    DMA_InitTypeDef DMA_InitStructure;

    // 1. 开启 DMA1 时钟
    RCC_AHBPeriphClockCmd(RCC_AHBPeriph_DMA1, ENABLE);

    // 2. 复位 DMA1 Channel7 (USART2_TX 专用通道)
    DMA_DeInit(DMA1_Channel7);

    // 3. 配置参数
    DMA_InitStructure.DMA_PeripheralBaseAddr = (uint32_t)&USART2->DR; // USART2 数据寄存器
    DMA_InitStructure.DMA_MemoryBaseAddr = (uint32_t)0;               // 发送时指定
    DMA_InitStructure.DMA_DIR = DMA_DIR_PeripheralDST;              // 内存 -> 外设
    DMA_InitStructure.DMA_BufferSize = 0;
    DMA_InitStructure.DMA_PeripheralInc = DMA_PeripheralInc_Disable;
    DMA_InitStructure.DMA_MemoryInc = DMA_MemoryInc_Enable;
    DMA_InitStructure.DMA_PeripheralDataSize = DMA_PeripheralDataSize_Byte;
    DMA_InitStructure.DMA_MemoryDataSize = DMA_MemoryDataSize_Byte;
    DMA_InitStructure.DMA_Mode = DMA_Mode_Normal;
    DMA_InitStructure.DMA_Priority = DMA_Priority_Medium;
    DMA_InitStructure.DMA_M2M = DMA_M2M_Disable;

    DMA_Init(DMA1_Channel7, &DMA_InitStructure);
}

/* --- USART2 初始化 --- */
void USART2_Init(void) {
    GPIO_InitTypeDef GPIO_InitStruct;
    USART_InitTypeDef USART_InitStruct;

    // 1. 开启时钟
    // 注意：USART2 在 APB1，GPIOA 在 APB2
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE); 
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_USART2, ENABLE); // 关键区别！

    // 2. GPIO 配置
    // PA2 - TX (复用推挽)
    GPIO_InitStruct.GPIO_Pin = GPIO_Pin_2;
    GPIO_InitStruct.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_InitStruct.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStruct);

    // PA3 - RX (浮空输入)
    GPIO_InitStruct.GPIO_Pin = GPIO_Pin_3;
    GPIO_InitStruct.GPIO_Mode = GPIO_Mode_IN_FLOATING;
    GPIO_Init(GPIOA, &GPIO_InitStruct);

    // 3. USART2 参数配置
    USART_InitStruct.USART_BaudRate = 115200;
    USART_InitStruct.USART_WordLength = USART_WordLength_8b;
    USART_InitStruct.USART_StopBits = USART_StopBits_1;
    USART_InitStruct.USART_Parity = USART_Parity_No;
    USART_InitStruct.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStruct.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    USART_Init(USART2, &USART_InitStruct);

    // 4. 初始化 DMA (Channel 7)
    USART2_DMA_Configuration();

    // 5. 开启 USART2 的 DMA 发送请求
    USART_DMACmd(USART2, USART_DMAReq_Tx, ENABLE);

    // 6. 使能 USART2
    USART_Cmd(USART2, ENABLE);
}

/* --- 新的 DMA 发送函数 --- */
void USART2_DMA_Transmit(uint8_t *pBuffer, uint16_t Size) {
    DMA_Cmd(DMA1_Channel7, DISABLE);             // 关闭 Channel 7
    DMA1_Channel7->CMAR = (uint32_t)pBuffer;     // 设置地址
    DMA1_Channel7->CNDTR = Size;                 // 设置长度
    DMA_Cmd(DMA1_Channel7, ENABLE);              // 启动
}

/* --- 修改后的按键初始化 (避开 PA2/PA3) --- */
void Button_GPIO_Init(void) {
    GPIO_InitTypeDef GPIO_InitStructure;
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);

    // 使用 PA0, PA1, PA4, PA6, PA7 作为新按键引脚
    // 避开了被串口占用的 PA2, PA3
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0 | GPIO_Pin_1 | GPIO_Pin_4 | GPIO_Pin_6 | GPIO_Pin_7;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IPD; 
    GPIO_Init(GPIOA, &GPIO_InitStructure);
}
// 简单的按键处理变量
static uint8_t joint_angle = 90; 
static uint32_t last_key_tick = 0;

/**
 * @brief 按键扫描任务
 */
void Key_Polling_Task(void) {
    if (GetTick() - last_key_tick < 50) return; // 50ms 扫描一次
    last_key_tick = GetTick();

    // 读取 PA0 (假设用于增加角度)
    if (GPIO_ReadInputDataBit(GPIOA, GPIO_Pin_0) == Bit_SET) {
        if (joint_angle < 180) joint_angle++;
        Remote_UpdateArmData(0, joint_angle); // 更新关节 0 数据
    }
    // 读取 PA1 (假设用于减小角度)
    else if (GPIO_ReadInputDataBit(GPIOA, GPIO_Pin_1) == Bit_SET) {
        if (joint_angle > 0) joint_angle--;
        Remote_UpdateArmData(0, joint_angle); // 更新关节 0 数据
    }
}
